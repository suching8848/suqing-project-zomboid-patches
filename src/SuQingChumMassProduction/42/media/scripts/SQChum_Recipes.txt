module Base
{
    /* Keep the vanilla knowledge ID used by books, professions and sand-ground
       actions. B42 appends same-ID IO; SQChum/Compatibility.lua removes exactly
       the known vanilla one-use input/output after native initialization. */
    craftRecipe MakeChum
    {
        timedAction = MakeChum,
        time = 120,
        NeedToBeLearn = true,
        Tags = InHandCraft;Fishing,
        category = Fishing,
        AutoLearnAny = Fishing:5,
        Tooltip = Tooltip_SQChum_Bag,
        inputs
        {
            /* A full vanilla bag has four uses. IsFull rejects partial bags.
               Native depletion supplies ONE empty bag; do not output another. */
            item 4 [Base.Sandbag] flags[IsFull],
            item 1 [*],
            -fluid 0.1 [Water;TaintedWater] mode:mixture,
        }
        outputs
        {
            item 5 Base.Chum,
        }
    }
